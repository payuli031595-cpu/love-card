    let isPlaying = false;


    async function startMusic() {

      try {

        await audio.play();

        isPlaying = true;

        toggleBtn.textContent = "⏸ 暫停音樂";

        tapToPlay.style.display = "none";

      } catch (e) {

        tapToPlay.style.display = "block";

        toggleBtn.textContent = "🎵 播放音樂";

      }

    }


    function pauseMusic() {

      audio.pause();

      isPlaying = false;

      toggleBtn.textContent = "🎵 播放音樂";

    }


    function toggleMusic() {

      if (isPlaying) pauseMusic();

      else startMusic();

    }
